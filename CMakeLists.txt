cmake_minimum_required(VERSION 3.25)
project(TopEnd VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TOPEND_BUILD_PLUGINS "Build iPlug2 AUv2/VST3/CLAP adapters" OFF)
function(topend_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endfunction()
add_library(topend-dsp STATIC src/dsp/TopEndProcessor.cpp)
set_target_properties(topend-dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(topend-dsp PUBLIC src)
topend_warnings(topend-dsp)
add_executable(topend-render tools/render.cpp)
target_link_libraries(topend-render PRIVATE topend-dsp)
topend_warnings(topend-render)
include(CTest)
if(BUILD_TESTING)
  add_executable(topend-dsp-tests tests/dsp/tests.cpp)
  target_link_libraries(topend-dsp-tests PRIVATE topend-dsp)
  topend_warnings(topend-dsp-tests)
  add_test(NAME dsp-and-state COMMAND topend-dsp-tests)
endif()
if(TOPEND_BUILD_PLUGINS)
  add_subdirectory(src/plugin)
  if(UNIX)
    add_executable(topend-clap-host tools/clap_host.cpp)
    target_include_directories(topend-clap-host PRIVATE external/clap/include)
    target_link_libraries(topend-clap-host PRIVATE ${CMAKE_DL_LIBS})
    topend_warnings(topend-clap-host)
  endif()
endif()
add_executable(topend-benchmark tests/performance/benchmark.cpp)
target_link_libraries(topend-benchmark PRIVATE topend-dsp)
if(APPLE)
  enable_language(OBJCXX)
  add_executable(topend-au-host tools/au_host.mm)
  target_compile_options(topend-au-host PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(topend-au-host PRIVATE "-framework AudioToolbox" "-framework Foundation" "-framework AppKit")
endif()
